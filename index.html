import { FBXLoader } from 'three/addons/loaders/FBXLoader.js';

const loader = new FBXLoader();
loader.load('../fbx/character.fbx', (character) => {

STEP 03

애니메이션 입히기 서비스의 심장

별도의 walk_ani.fbx에서 동작만 꺼내 캐릭터에 적용·재생.

핵심 원리

  1. AnimationMixer(character) — 캐릭터 전용 재생기 생성
  2. walk_ani.fbx의 animations[0] = 동작 데이터(Clip)
  3. mixer.clipAction(clip).play() — 연결 후 재생
  4. 매 프레임 mixer.update(delta) — 시간 흘려보내기
  5. 전제: 두 FBX의 뼈 이름이 같아야 함

핵심 코드


const clock = new THREE.Clock();
let mixer;

loader.load('../fbx/character.fbx', (character) => {
  scene.add(character);
  mixer = new THREE.AnimationMixer(character);      // ① 재생기

  loader.load('../fbx/walk_ani.fbx', (anim) => {
    const clip = anim.animations[0];                // ② 동작 데이터
    mixer.clipAction(clip).play();                  // ③ 재생
  });
});

// 렌더 루프
function animate() {
  requestAnimationFrame(animate);
  if (mixer) mixer.update(clock.getDelta());        // ④ 진행
  renderer.render(scene, camera);
}

③ 검증 / ④ 오류

  • 캐릭터가 T자에서 벗어나 실제로 움직인다
  • 동작이 찌그러지지 않는다 (찌그러지면 뼈 이름 불일치)

· 안 움직임 → mixer.update() 누락 / play() 누락

· 몸이 뒤틀림 → 두 FBX 뼈 이름 불일치

"애니메이션을 입히니 캐릭터가 뒤틀려. 두 FBX의 뼈 이름이 다를 때 맞추는(리타게팅) 방법을 알려줘."

"여러 애니메이션 FBX를 배열로 관리하며 제어 패널을 만들어줘. ① 애니메이션마다 전환 버튼(0.4초 크로스페이드), ② 재생/일시정지 토글, ③ 속도 슬라이더(0~2배), ④ 재생 위치를 보여주고 드래그로 이동하는 타임라인. 외부 UI 라이브러리 없이 순수 HTML/CSS로." 

// 부드러운 전환 — 이전 동작에서 새 동작으로 0.4초 크로스페이드

파일 불러오기 인터페이스

경로를 코드에 박지 않고, 사용자가 직접 FBX를 골라 읽게 한다.

버튼 선택 · 드래그&드롭 · 캐릭터/애니메이션 자동 인식

핵심 개념

  1. 서버 URL이 아니므로 load() 대신 FileReader → loader.parse()
  2. 자동 분류: 스킨 메시가 있으면 캐릭터, 없으면 애니메이션
  3. 불러온 클립을 보관 → 캐릭터를 바꿔도 다시 적용
  4. 모델 크기가 제각각 → 카메라 자동 맞춤

const reader = new FileReader();
reader.onload = (e) => {
  const obj = loader.parse(e.target.result, ''); // ArrayBuffer → 3D 객체
  obj.traverse(o => { if (o.isSkinnedMesh) isCharacter = true; });
};
reader.readAsArrayBuffer(file);

왜 glTF/GLB로 변환하나

FBX는 웹 배포에 무겁다

이 강의에서 실제로 변환한 결과 ↓

FBXGLB
character49 MB29 MB
walk_ani34 MB29 MB
합계83 MB58 MB (−30%)

모캡 애니메이션은 프레임마다 값이 달라 더 줄이려면 meshopt/Draco 압축 필요

FBX vs glTF/GLB

FBXglTF / GLB
표준Autodesk 독점웹 개방 표준(Khronos)
용량큼작음 (압축 가능)
로딩느림·무거운 파서빠름 (브라우저 친화)
별칭—"웹의 JPEG" 라 불림

GLB = glTF(장면·애니메이션) + 텍스처를 한 파일로 묶은 바이너리 형식

변환 방법 (이 강의에서 실제 사용)

① fbx2gltf (npm) — 설치 후 짧은 스크립트로 변환

// npm i fbx2gltf  후 실행
const convert = require('fbx2gltf');
convert('fbx/character.fbx', 'glb/character.glb', ['--binary'])
  .then(out => console.log('완료:', out));

② Blender — FBX Import → glTF 2.0(.glb) Export (GUI, 무료)

③ gltf-transform로 추가 최적화 (더 줄이려면 meshopt/Draco)

npx @gltf-transform/cli optimize in.glb out.glb --compress meshopt

"FBX를 웹용 GLB로 변환하는 방법을 알려줘. Windows에 Blender가 없을 때 npm으로 하는 법도."

코드에서 GLB 쓰기

FBXLoader 대신 GLTFLoader 사용 — 사용법은 거의 동일


import { GLTFLoader } from 'three/addons/loaders/GLTFLoader.js';

const loader = new GLTFLoader();
loader.load('../glb/character.glb', (gltf) => {
  scene.add(gltf.scene);                              // 캐릭터
  const mixer = new THREE.AnimationMixer(gltf.scene);
  // 애니메이션은 gltf.animations 배열에 들어 있다
  mixer.clipAction(gltf.animations[0]).play();
});

캐릭터·애니메이션을 한 GLB로 합치거나, 파일을 나눠 관리할 수도 있다.

GitHub이란?

Git = 코드의 버전 관리 도구. 언제 무엇을 바꿨는지 기록·되돌리기.

GitHub = 그 코드를 인터넷에 저장·공유하는 공간(클라우드 저장소).

  • Repository(저장소) — 프로젝트 하나가 담기는 폴더
  • Commit — "이 시점의 상태"를 저장한 스냅샷
  • Push — 내 컴퓨터의 커밋을 GitHub로 올리기

비유: 구글 드라이브 + '되돌리기 기록'이 붙은 코드 전용 창고

GitHub Pages란?

저장소에 올린 HTML/CSS/JS 파일을 그대로 웹사이트로 무료 배포해 주는 기능. 별도 서버·도메인·비용이 필요 없다.
  • 저장소의 파일이 곧 웹페이지가 된다 (정적 호스팅)
  • 주소 규칙: https://<아이디>.github.io/<저장소>/
  • Push 하면 자동으로 사이트가 갱신된다

우리 서비스는 순수 HTML+Three.js(정적)라 Pages에 완벽히 맞다.

① 코드 올리기 (Push)


git init
git add .
git commit -m "FBX 3D 웹서비스"
git branch -M main
git remote add origin https://github.com/dknife/DeptTalk2026.git
git push -u origin main

처음 push 시 GitHub 로그인(브라우저 인증) 창이 뜬다. Git이 낯설면: 저장소 → Add file → Upload files 드래그 업로드도 가능.

② Pages 켜기

  1. 저장소 DeptTalk2026 → Settings → Pages
  2. Source: Deploy from a branch → main / /(root) → Save
  3. 1~2분 뒤 공개 주소 생성 ↓

🎞 슬라이드: https://dknife.github.io/DeptTalk2026/

🕹 실습: https://dknife.github.io/DeptTalk2026/examples/

배포 검증 포인트

  • 경로 대소문자 — GitHub은 구분함 (Fbx ≠ fbx)
  • 상대경로 — ../fbx/... 가 실제 구조와 맞는지
  • 모델 용량 — 너무 크면 로딩이 느림
  • 공개 주소로 접속해 로컬과 동일하게 동작하는지

"GitHub Pages에서 모델 파일 404가 떠. 로컬에선 됐어. 원인 후보를 알려줘."

직접 따라 만들기

AI에게 준 "지시 순서" 그대로 재현하기

이 강의 자료·서비스·배포는 코드를 직접 타이핑하지 않고,
아래 7개의 지시를 순서대로 AI(Claude)에게 요청해 완성했다.

지시 순서 한눈에 보기

#AI에게 준 지시(요지)산출물
1FBX 캐릭터+애니메이션을 WebGL로 재생·제어하는 강의 자료 (순수 Three.js·CDN, 설명+코드)실습 01~04 · 개념 정리
2플랫폼은 GitHub, 구축 과정을 차례로 — AI 활용 역량 교육이 목적단계별 프롬프트·검증 구조
3모델은 fbx 폴더에 있음, 강의노트는 HTML 프리젠테이션으로이 슬라이드(reveal.js)
4예제에 파일 불러오기 UI 추가 (캐릭터·애니메이션 직접 로드)실습 05
5FBX를 GLB로 변환하고 그 필요성을 노트에 기록glb/ · 변환 슬라이드
6examples를 순서대로 여는 index.html 런처examples/index.html
7GitHub 푸시 + Pages 배포, GitHub·Pages 개념도 슬라이드에라이브 서비스 공개

핵심: "구체적으로 지시 → 결과 검증 → 다음 지시"의 반복

지시 1~3 · 기획과 형식 정하기

"캐릭터 FBX와 애니메이션 FBX를 WebGL로 재생·제어하는 웹서비스 강의 자료를 만들어줘. 순수 Three.js를 CDN으로 쓰고, 설명 문서와 실행 코드를 함께. 목적은 AI 활용 역량 교육이야."
"모델은 fbx 폴더에 있어. 강의노트는 별도 HTML 프리젠테이션으로 만들어줘."

💡 배운 점 — 처음에 결과물의 형식(문서? 슬라이드?)과 기술 스택(순수 Three.js?)을 분명히 정해 주면 AI가 헤매지 않는다. AI가 되물으면 골라 주면 된다.

지시 4~5 · 기능 추가와 최적화

"예제를 수정해 파일을 직접 불러오는 인터페이스를 추가하고 캐릭터·애니메이션을 읽게 해줘."
"FBX를 GLB로 변환하고, 이 과정이 필요한 이유를 강의노트에도 기록해줘."

→ AI가 실제로 한 일: fbx2gltf 설치·변환(83MB→58MB), 변환 슬라이드 작성, FileReader+parse()로 업로드 UI 구현.

💡 배운 점 — "무엇을" 뿐 아니라 "왜 필요한지 문서에도 남겨줘"처럼 산출물의 범위까지 지시하면 완성도가 올라간다.

지시 6~7 · 정리와 배포

"examples를 순서대로 실행하는 index.html 런처를 만들어줘."
"github.com/dknife/DeptTalk2026 에 푸시해서 Pages로 배포하고, GitHub·Pages 개념과 올리는 방법도 슬라이드에 넣어줘."

→ AI가 실제 실행한 절차:


git init; git add .; git commit -m "..."
git remote add origin https://github.com/dknife/DeptTalk2026.git
git push -u origin main          # → Pages 자동 빌드 → 사이트 공개

⚠ 이때 AI가 스스로 잡아낸 함정: FBX/GLB 바이너리가 손상되지 않도록 .gitattributes로 처리(검증까지). — 검증은 사람도 AI도 반드시 해야 한다.

여러분이 재현하려면

  1. fbx/ 폴더에 character.fbx·애니메이션 FBX 준비 (Mixamo)
  2. 위 지시 1~7을 순서대로 AI에게 요청 — 한 번에 하나씩
  3. 각 단계마다 브라우저로 열어 직접 검증 (안 되면 오류를 근거로 다시 요청)
  4. GitHub 저장소 만들고 git push → Settings → Pages 켜기
🎯 이 강의의 결론이 그대로 방법이 된다: 쪼개고 · 지시하고 · 검증하고 · 반복하라.

마무리 — 무엇을 배웠나

  1. 목표를 4단계로 분해했다
  2. 각 단계를 AI에게 맥락과 함께 요청했다
  3. 결과를 체크리스트로 검증했다
  4. 안 되면 오류를 근거로 다시 요청했다
  5. GitHub Pages로 공개했다

강의 결론

AI 시대의 핵심 능력은 코드 암기가 아니라

쪼개고 · 잘 요청하고 · 검증하고 · 반복 개선하기

이 4가지는 3D 뷰어뿐 아니라 모든 AI 협업 프로젝트에 그대로 적용된다.
